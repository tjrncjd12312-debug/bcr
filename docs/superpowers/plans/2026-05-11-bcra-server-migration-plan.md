# BCR Server Migration to Supabase — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the expired `bcra.store` Spring server with a Supabase (Postgres + Edge Functions) backend that the existing newbcr Tauri client can use with only a base-URL change.

**Architecture:** Single Supabase Edge Function named `api` with internal path-based routing, talking to Postgres via the service-role key. Custom HS256 JWT (not Supabase Auth) preserves the existing client API contract verbatim. Predict endpoints return random Banker/Player/skip stubs. Duplicate-login is enforced by a UNIQUE(user_id) constraint on an `active_sessions` table.

**Tech Stack:**
- **Server (new):** Supabase (Postgres 15, Edge Functions in Deno/TypeScript), `djwt`, `bcryptjs`
- **Server tests:** Deno built-in test runner + `fetch` against `supabase functions serve` (local)
- **Client (modified):** Rust (Tauri backend) — 2 small edits in existing files

**Spec:** `docs/superpowers/specs/2026-05-11-bcra-server-migration-design.md`

```ts
// supabase/functions/api/jwt.ts
import { jwtCreate, jwtVerify } from "./deps.ts";

let cachedKey: CryptoKey | null = null;

async function getKey(): Promise<CryptoKey> {
  if (cachedKey) return cachedKey;
  const secret = Deno.env.get("JWT_SECRET");
  if (!secret) throw new Error("JWT_SECRET not configured");
  cachedKey = await crypto.subtle.importKey(
    "raw",
    new TextEncoder().encode(secret),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign", "verify"],
  );
  return cachedKey;
}

export interface TokenPayload {
  sub: string;       // user_id as string
  jti: string;       // session id
  username: string;
  exp: number;       // unix seconds
  iat: number;
}

export async function sign(payload: Omit<TokenPayload, "iat">): Promise<string> {
  const key = await getKey();
  return await jwtCreate(
    { alg: "HS256", typ: "JWT" },
    { ...payload, iat: Math.floor(Date.now() / 1000) },
    key,
  );
}

export type VerifyResult =
  | { ok: true; payload: TokenPayload }
  | { ok: false; reason: "expired" | "invalid" };

export async function verify(token: string): Promise<VerifyResult> {
  try {
    const key = await getKey();
    const payload = await jwtVerify(token, key) as unknown as TokenPayload;
    return { ok: true, payload };
  } catch (e) {
    const msg = (e as Error).message ?? "";
    if (msg.includes("exp") || msg.toLowerCase().includes("expired")) {
      return { ok: false, reason: "expired" };
    }
    return { ok: false, reason: "invalid" };
  }
}

export function extractBearer(req: Request): string | null {
  const auth = req.headers.get("Authorization") ?? "";
  const m = auth.match(/^Bearer\s+(.+)$/);
  return m ? m[1] : null;
}
```

```ts
// supabase/functions/api/handlers/login.ts
import { bcrypt } from "../deps.ts";
import { getDb } from "../db.ts";
import { sign } from "../jwt.ts";
import { json } from "../responses.ts";

interface LoginRequest {
  username?: string;
  password?: string;
}

export async function handleLogin(req: Request): Promise<Response> {
  let body: LoginRequest;
  try { body = await req.json(); } catch {
    return json({ success: false, message: "잘못된 요청 형식입니다" }, 400);
  }
  const { username, password } = body;
  if (!username || !password) {
    return json({ success: false, message: "아이디와 비밀번호를 입력하세요" }, 400);
  }

  const db = getDb();

  // 1. Fetch user
  const { data: user, error: userErr } = await db
    .from("users")
    .select("id, username, password_hash, expiry_at, active")
    .eq("username", username)
    .maybeSingle();

  if (userErr) {
    console.error("DB error:", userErr);
    return json({ success: false, message: "서버 오류" }, 500);
  }
  if (!user || !user.active) {
    return json({ success: false, message: "아이디 또는 비밀번호가 올바르지 않습니다" }, 401);
  }

  // 2. Password check
  const ok = await bcrypt.compare(password, user.password_hash);
  if (!ok) {
    return json({ success: false, message: "아이디 또는 비밀번호가 올바르지 않습니다" }, 401);
  }

  // 3. Expiry check
  const expiryEpochMs = new Date(user.expiry_at).getTime();
  const nowMs = Date.now();
  if (expiryEpochMs <= nowMs) {
    return json({ success: false, message: "이용권이 만료되었습니다" }, 401);
  }
  const seconds = Math.floor((expiryEpochMs - nowMs) / 1000);

  // 4. Replace any existing session for this user
  const jti = crypto.randomUUID();
  const { error: delErr } = await db.from("active_sessions").delete().eq("user_id", user.id);
  if (delErr) { console.error("delete session err:", delErr); }

  const { error: insErr } = await db.from("active_sessions").insert({
    jti,
    user_id: user.id,
    expires_at: new Date(expiryEpochMs).toISOString(),
  });
  if (insErr) {
    console.error("insert session err:", insErr);
    return json({ success: false, message: "세션 생성 실패" }, 500);
  }

  // 5. Sign JWT
  const token = await sign({
    sub: String(user.id),
    jti,
    username: user.username,
    exp: Math.floor(expiryEpochMs / 1000),
  });

  // 6. Latest active notice
  const { data: notice } = await db
    .from("notices")
    .select("id, title, content, created_at")
    .eq("active", true)
    .order("created_at", { ascending: false })
    .limit(1)
    .maybeSingle();

  return json({
    success: true,
    token,
    userId: String(user.id),
    username: user.username,
    seconds,
    notice: notice ? {
      id: notice.id,
      title: notice.title,
      content: notice.content,
      regDate: notice.created_at,
    } : null,
    message: null,
  });
}
```

```ts
// supabase/functions/api/handlers/validateToken.ts
import { getDb } from "../db.ts";
import { verify, extractBearer } from "../jwt.ts";
import { json } from "../responses.ts";

export async function handleValidateToken(req: Request): Promise<Response> {
  const token = extractBearer(req);
  if (!token) return json({ success: false, message: "인증 토큰 없음" }, 401);

  const v = await verify(token);
  if (!v.ok) {
    if (v.reason === "expired") {
      return json({ valid: false, expired: true, remainingSeconds: 0 });
    }
    return json({ success: false, message: "유효하지 않은 토큰" }, 401);
  }

  const { jti, sub } = v.payload;
  const db = getDb();

  // Check session row exists (= this jti is still the current session)
  const { data: session } = await db
    .from("active_sessions")
    .select("jti, user_id")
    .eq("jti", jti)
    .maybeSingle();

  if (!session) {
    // 🎯 Duplicate-login: someone else logged in and deleted/replaced our row
    return json({ valid: false, expired: false });
  }

  // Fetch user expiry
  const { data: user } = await db
    .from("users")
    .select("expiry_at")
    .eq("id", sub)
    .maybeSingle();

  if (!user) {
    return json({ valid: false, expired: false });
  }

  const expiryEpochMs = new Date(user.expiry_at).getTime();
  const nowMs = Date.now();
  if (expiryEpochMs <= nowMs) {
    await db.from("active_sessions").delete().eq("jti", jti);
    return json({ valid: false, expired: true, remainingSeconds: 0 });
  }

  const remainingSeconds = Math.floor((expiryEpochMs - nowMs) / 1000);

  // Touch last_validated_at (best-effort; ignore errors)
  await db.from("active_sessions").update({ last_validated_at: new Date().toISOString() }).eq("jti", jti);

  return json({ valid: true, expired: false, remainingSeconds });
}
```

```ts
// supabase/functions/api/handlers/versionCheck.ts
import { getDb } from "../db.ts";
import { json } from "../responses.ts";

interface VersionRequest { clientVersion?: string; platform?: string }

function parseSemver(v: string): number[] {
  return v.split(".").map((p) => parseInt(p, 10) || 0);
}
function semverGte(a: string, b: string): boolean {
  const [aMaj, aMin, aPat] = parseSemver(a);
  const [bMaj, bMin, bPat] = parseSemver(b);
  if (aMaj !== bMaj) return aMaj > bMaj;
  if (aMin !== bMin) return aMin > bMin;
  return aPat >= bPat;
}

export async function handleVersionCheck(req: Request): Promise<Response> {
  let body: VersionRequest = {};
  try { body = await req.json(); } catch { /* tolerate */ }
  const platform = body.platform ?? "desktop";
  const clientVersion = body.clientVersion ?? "0.0.0";

  const db = getDb();
  const { data: row } = await db
    .from("app_versions")
    .select("required_version, download_url, file_name, message")
    .eq("platform", platform)
    .maybeSingle();

  if (!row) {
    // Default: assume compatible if no policy exists
    return json({
      requiredVersion: "0.0.0",
      compatible: true,
      message: null,
      downloadUrl: null,
      fileName: null,
    });
  }

  const compatible = semverGte(clientVersion, row.required_version);
  return json({
    requiredVersion: row.required_version,
    compatible,
    message: row.message,
    downloadUrl: row.download_url,
    fileName: row.file_name,
  });
}
```
